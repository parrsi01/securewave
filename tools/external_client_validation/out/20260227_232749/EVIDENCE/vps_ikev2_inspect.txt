# /lib/systemd/system/strongswan-starter.service
[Unit]
Description=strongSwan IPsec IKEv1/IKEv2 daemon using ipsec.conf
After=network-online.target

[Service]
ExecStart=/usr/sbin/ipsec start --nofork
ExecReload=/usr/sbin/ipsec reload
Restart=on-abnormal

[Install]
WantedBy=multi-user.target

config setup
  uniqueids=no

conn securewave-ikev2-eap
  auto=add
  type=tunnel
  keyexchange=ikev2
  fragmentation=yes
  forceencaps=yes
  dpdaction=clear
  dpddelay=300s
  rekey=no

  left=%any
  leftid=138.199.204.139
  leftcert=server-cert.pem
  leftsendcert=always
  leftsubnet=0.0.0.0/0

  right=%any
  rightid=%any
  rightauth=eap-mschapv2
  rightsourceip=10.10.0.10-10.10.0.250
  rightdns=1.1.1.1,1.0.0.1
  rightsendcert=never
  eap_identity=%identity

  ike=aes256gcm16-prfsha256-ecp256,aes128gcm16-prfsha256-ecp256,aes256-sha256-modp2048!
  esp=aes256gcm16-ecp256,aes128gcm16-ecp256,aes256-sha256!
