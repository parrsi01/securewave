=== script ===
#!/usr/bin/env bash
set -euo pipefail
OUT_IF="$(ip -4 route show default | awk '{print $5; exit}')"
[ -n "$OUT_IF" ] || OUT_IF=eth0

# WireGuard
iptables -t nat -C POSTROUTING -s 10.8.0.0/24 -o "$OUT_IF" -j MASQUERADE >/dev/null 2>&1 || iptables -t nat -A POSTROUTING -s 10.8.0.0/24 -o "$OUT_IF" -j MASQUERADE
iptables -C FORWARD -i wg0 -o "$OUT_IF" -j ACCEPT >/dev/null 2>&1 || iptables -A FORWARD -i wg0 -o "$OUT_IF" -j ACCEPT
iptables -C FORWARD -i "$OUT_IF" -o wg0 -m conntrack --ctstate RELATED,ESTABLISHED -j ACCEPT >/dev/null 2>&1 || iptables -A FORWARD -i "$OUT_IF" -o wg0 -m conntrack --ctstate RELATED,ESTABLISHED -j ACCEPT

# OpenVPN
iptables -t nat -C POSTROUTING -s 10.9.0.0/24 -o "$OUT_IF" -j MASQUERADE >/dev/null 2>&1 || iptables -t nat -A POSTROUTING -s 10.9.0.0/24 -o "$OUT_IF" -j MASQUERADE
iptables -C FORWARD -i tun0 -o "$OUT_IF" -j ACCEPT >/dev/null 2>&1 || iptables -A FORWARD -i tun0 -o "$OUT_IF" -j ACCEPT
iptables -C FORWARD -i "$OUT_IF" -o tun0 -m conntrack --ctstate RELATED,ESTABLISHED -j ACCEPT >/dev/null 2>&1 || iptables -A FORWARD -i "$OUT_IF" -o tun0 -m conntrack --ctstate RELATED,ESTABLISHED -j ACCEPT

# IKEv2
iptables -t nat -C POSTROUTING -s 10.10.0.0/24 -o "$OUT_IF" -j MASQUERADE >/dev/null 2>&1 || iptables -t nat -A POSTROUTING -s 10.10.0.0/24 -o "$OUT_IF" -j MASQUERADE
iptables -C FORWARD -m policy --pol ipsec --dir in -j ACCEPT >/dev/null 2>&1 || iptables -A FORWARD -m policy --pol ipsec --dir in -j ACCEPT
iptables -C FORWARD -m policy --pol ipsec --dir out -j ACCEPT >/dev/null 2>&1 || iptables -A FORWARD -m policy --pol ipsec --dir out -j ACCEPT

=== unit ===

[Service]
